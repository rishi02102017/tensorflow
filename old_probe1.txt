def T(name, fn):
    try:
        r = fn()
        if isinstance(r, (tf.Tensor, tf.RaggedTensor, tf.SparseTensor)):
            r = r.numpy() if isinstance(r, tf.Tensor) else r
        s = repr(r).replace("\n", " ")
        print(f"[OK ] {name}: {s[:300]}")
    except Exception as e:
        print(f"[ERR] {name}: {type(e).__name__}: {str(e).splitlines()[0][:300] if str(e) else ''}")

def CMP(name, tf_fn, np_fn):
    try:
        a = tf_fn()
        a = np.asarray(a)
    except Exception as e:
        a = e
    try:
        b = np.asarray(np_fn())
    except Exception as e:
        b = e
    if isinstance(a, Exception) or isinstance(b, Exception):
        if isinstance(a, Exception) and isinstance(b, Exception):
            print(f"[SAME-ERR] {name}")
        else:
            print(f"[MISMATCH] {name}: tf={type(a).__name__ if isinstance(a,Exception) else a.shape}:{str(a)[:200] if isinstance(a,Exception) else ''} np={type(b).__name__ if isinstance(b,Exception) else b.shape}:{str(b)[:200] if isinstance(b,Exception) else ''}")
        return
    ok = a.shape == b.shape and np.allclose(a.astype(np.float64) if a.dtype!=object else a, b.astype(np.float64) if b.dtype!=object else b, equal_nan=True)
    dt = "" if a.dtype == b.dtype else f" dtype tf={a.dtype} np={b.dtype}"
    print(f"[{'match' if ok else 'MISMATCH'}] {name}: tf={a.shape} np={b.shape}{dt}" + ("" if ok else f" tf={str(a.tolist())[:150]} np={str(b.tolist())[:150]}"))
