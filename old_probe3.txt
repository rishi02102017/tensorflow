def T(name, fn, tb=False):
    try:
        r = fn()
        if hasattr(r, 'numpy'): r = r.numpy()
        print(f"[OK ] {name}: {repr(r)[:300]}".replace("\n", " "))
    except Exception as e:
        msg = str(e)
        print(f"[ERR] {name}: {type(e).__name__}: {msg[-700:]}".replace("\n", " | "))
        if tb:
            traceback.print_exc(limit=-6)
